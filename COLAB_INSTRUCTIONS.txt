def hex2rgba(h):
    h = h.lstrip('#'); return tuple(int(h[i:i+2],16)/255 for i in (0,2,4)) + (1,)

def get_lang():
    try:
        loc = locale.getdefaultlocale()[0]
        if loc and loc.startswith('es'): return 'es'
        if loc and loc.startswith('ro'): return 'ro'
    except: pass
    return 'en'

class YtMp3App(App):
    def build(self):
        self.lang = get_lang(); self.t = T[self.lang]; self.title = self.t["title"]
        self.downloading = False; self.queue = []
        self.download_dir = os.path.expanduser('~/Downloads/MP3')
        os.makedirs(self.download_dir, exist_ok=True)
        root = BoxLayout(orientation='vertical', padding=12, spacing=8)

        self.dl_btn = Button(text=self.t["download"], background_color=hex2rgba(GREEN), color=(0,0,0,1), bold=True, font_size='13sp', size_hint_y=None, height=44)
        self.dl_btn.bind(on_press=self.start_download); root.add_widget(self.dl_btn)
        root.add_widget(Label(text=self.t["dedication"], font_size='13sp', bold=True, color=hex2rgba(NEON), size_hint_y=None, height=28))
        return root
    def add_urls(self, *a):
        text = self.url_input.text.strip()
        if not text: return
        urls = [l.strip() for l in text.split('\n') if re.search(r'(youtube\.com|youtu\.be)', l.strip())]
        if not urls: self.status_lbl.text = "No valid links"; self.status_lbl.color = hex2rgba(RED); return
        for url in urls:
            self.queue.append({'url':url,'status':'pending','title':''})
            self.queue_box.add_widget(Label(text=url[:60]+'...' if len(url)>60 else url, font_size='11sp', color=hex2rgba(TXT), size_hint_y=None, height=30))
        self.url_input.text=''; self._update_count()
    def clear_queue(self, *a):
        if self.downloading: return
        self.queue.clear(); self.queue_box.clear_widgets(); self._update_count()
    def _update_count(self):
        n=len(self.queue); p=sum(1 for q in self.queue if q['status']=='pending')
        self.count_lbl.text = self.t["count"].format(n=n,p=p)
    def start_download(self, *a):
        if not [q for q in self.queue if q['status']=='pending']: return
        self.downloading=True; self.dl_btn.disabled=True; self.add_btn.disabled=True; self.clear_btn.disabled=True; self.progress.value=0
        threading.Thread(target=self._download_all, daemon=True).start()
    def _download_all(self):
        import yt_dlp
        pending=[q for q in self.queue if q['status']=='pending']; total=len(pending); ok=err=0
        for i,item in enumerate(pending):
            item['status']='downloading'
            Clock.schedule_once(lambda dt,t=i: self.status_lbl setattr__('text', self.t["downloading"]), 0)
            def hook(d,idx=i):
                if d['status']=='downloading':
                    try:
                        pct=float(re.search(r'([\d.]+)',d.get('_percent_str','0')).group(1))
                        Clock.schedule_once(lambda dt,v=(idx*100+pct)/total: setattr(self.progress,'value',v), 0)
                    except: pass
                elif d['status']=='finished':
                    Clock.schedule_once(lambda dt: setattr(self.status_lbl,'text',self.t["converting"]), 0)
            opts={'format':'bestaudio/best','outtmpl':os.path.join(self.download_dir,'%(title)s.%(ext)s'),'progress_hooks':[hook],'postprocessors':[{'key':'FFmpegExtractAudio','preferredcodec':'mp3','preferredquality':'0'}],'quiet':True,'no_warnings':True}
            try:
                with yt_dlp.YoutubeDL(opts) as ydl:
                    info=ydl.extract_info(item['url'],download=False); item['title']=info.get('title','?'); ydl.download([item['url']])
                item['status']='ok'; ok+=1
            except: item['status']='error'; err+=1
            Clock.schedule_once(lambda dt,v=(i+1)/total*100: setattr(self.progress,'value',v), 0)
        Clock.schedule_once(lambda dt: self._finish(ok,err), 0)
    def _finish(self,ok,err):
        self.downloading=False; self.dl_btn.disabled=False; self.add_btn.disabled=False; self.clear_btn.disabled=False
        self.progress.value=100; self.status_lbl.text=self.t["completed"].format(ok=ok,err=err); self.status_lbl.color=hex2rgba(GREEN)
        self._update_count()
